async function getImportableLibrary(libraryId, userId) {
  const { data: library, error: libraryError } = await supabase
    .from("libraries")
    .select("id, user_id, name, description, is_public, share_on_profile, created_at")
    .eq("id", libraryId)
    .maybeSingle();

  if (libraryError) throw libraryError;

  const canAccess =
    library &&
    (library.is_public || String(library.user_id) === String(userId));

  return canAccess ? library : null;
}

Phân tích:

- Hàm lấy thư viện trực tiếp từ bảng "libraries".
- Một thư viện được phép import khi:
  + Thư viện là Public; hoặc
  + Người đang đăng nhập chính là chủ sở hữu.
- Nếu thư viện không tồn tại hoặc người dùng không có quyền, hàm trả về null.
- Việc trả về cùng một kết quả null cho cả "không tồn tại" và "không có quyền"
  giúp tránh làm lộ sự tồn tại của thư viện Private cho người lạ.


2. Hàm lấy các tài liệu có thể import

async function getImportableDocuments(libraryId) {
  const { data, error } = await supabase
    .from("documents")
    .select(
      "id, title, file_url, file_size_bytes, status, ai_reject_reason, created_at",
    )
    .eq("library_id", libraryId)
    .eq("status", "APPROVED")
    .is("deleted_at", null)
    .order("created_at", { ascending: true });

  if (error) throw error;
  return data || [];
}

Phân tích:

- Chỉ sao chép tài liệu đã được duyệt với status = "APPROVED".
- Không sao chép tài liệu đã bị xóa mềm, tức deleted_at phải là null.
- Tài liệu bị từ chối, đang chờ duyệt hoặc bị gắn cờ nhạy cảm sẽ không được import.
- Thứ tự created_at tăng dần giúp giữ thứ tự tương đối của tài liệu gốc.


3. API xem trước thư viện trước khi import

exports.previewLibraryImport = async (req, res) => {
  try {
    const { libraryId } = req.params;
    const userID = req.user.id;
    const library = await getImportableLibrary(libraryId, userID);

    if (!library) {
      return res.status(404).json({
        status: "error",
        message: "Library not found or is not available for import.",
      });
    }

    const documents = await getImportableDocuments(libraryId);
    const totalBytes = documents.reduce(
      (total, document) => total + (Number(document.file_size_bytes) || 0),
      0,
    );

    return res.status(200).json({
      status: "success",
      data: {
        id: library.id,
        name: library.name,
        description: library.description || "",
        visibility: library.is_public ? "public" : "private",
        documents: documents.length,
        folders: 0,
        totalBytes,
      },
    });
  } catch (error) {
    console.error("Preview library import error:", error);
    return res.status(500).json({
      status: "error",
      message: "Could not inspect this library for import.",
      error: error.message,
    });
  }
};

Phân tích:

- API này được nút "Check link" sử dụng.
- Nó xác minh thư viện từ database thay vì localStorage.
- Dữ liệu trả về gồm tên, mô tả, visibility, số tài liệu và tổng dung lượng.
- folders hiện trả về 0 vì cấu trúc thư mục của LibraryPage vẫn đang được lưu cục bộ,
  chưa có bảng folder tương ứng trong backend.


4. API thực hiện import

exports.importLibrary = async (req, res) => {
  const copiedStoragePaths = [];
  const insertedDocumentIds = [];
  let importedLibraryId = null;

  try {
    const { libraryId } = req.params;
    const userID = req.user.id;
    const name = String(req.body?.name || "").trim();
    const description = String(req.body?.description || "").trim();

    if (!name) {
      return res.status(400).json({
        status: "error",
        message: "Library name is required.",
      });
    }

    if (name.length > 50 || description.length > 350) {
      return res.status(400).json({
        status: "error",
        message: "Library name or description exceeds the allowed length.",
      });
    }

    const sourceLibrary = await getImportableLibrary(libraryId, userID);
    if (!sourceLibrary) {
      return res.status(404).json({
        status: "error",
        message: "Library not found or is not available for import.",
      });
    }

    const { data: duplicateLibrary, error: duplicateError } = await supabase
      .from("libraries")
      .select("id")
      .eq("user_id", userID)
      .ilike("name", name)
      .maybeSingle();

    if (duplicateError) throw duplicateError;
    if (duplicateLibrary) {
      return res.status(409).json({
        status: "error",
        code: "DUPLICATE_LIBRARY_NAME",
        message: `You already have a library named "${name}".`,
      });
    }

    const sourceDocuments = await getImportableDocuments(libraryId);
    const incomingBytes = sourceDocuments.reduce(
      (total, document) => total + (Number(document.file_size_bytes) || 0),
      0,
    );
    const { data: existingDocuments, error: storageError } = await supabase
      .from("documents")
      .select("file_size_bytes")
      .eq("uploader_id", userID)
      .not("library_id", "is", null)
      .is("deleted_at", null);

    if (storageError) throw storageError;
    const currentBytes = (existingDocuments || []).reduce(
      (total, document) => total + (Number(document.file_size_bytes) || 0),
      0,
    );

    if (currentBytes + incomingBytes > 50 * 1024 * 1024) {
      return res.status(400).json({
        status: "error",
        code: "USER_LIBRARY_STORAGE_LIMIT_EXCEEDED",
        message:
          "Importing this library would exceed your shared 50MB library storage limit.",
      });
    }

    const { data: importedLibrary, error: libraryInsertError } = await supabase
      .from("libraries")
      .insert({
        user_id: userID,
        name,
        description: description || sourceLibrary.description || "",
        is_public: false,
        share_on_profile: false,
      })
      .select()
      .single();

    if (libraryInsertError) throw libraryInsertError;
    importedLibraryId = importedLibrary.id;

    const sourceDocumentIds = sourceDocuments.map((document) => document.id);
    let sourceChunks = [];
    let sourceTags = [];

    if (sourceDocumentIds.length > 0) {
      const [
        { data: chunks, error: chunksError },
        { data: tags, error: tagsError },
      ] = await Promise.all([
        supabase
          .from("document_chunks")
          .select("document_id, chunk_index, content, embedding")
          .in("document_id", sourceDocumentIds),
        supabase
          .from("document_tags")
          .select("document_id, tag_id")
          .in("document_id", sourceDocumentIds),
      ]);

      if (chunksError) throw chunksError;
      if (tagsError) throw tagsError;
      sourceChunks = chunks || [];
      sourceTags = tags || [];
    }

    for (const sourceDocument of sourceDocuments) {
      const targetStoragePath =
        `${userID}/${Date.now()}-${crypto.randomUUID()}-` +
        sanitizeFileName(sourceDocument.title);
      const { error: copyError } = await supabase.storage
        .from(BUCKET)
        .copy(sourceDocument.file_url, targetStoragePath);

      if (copyError) throw copyError;
      copiedStoragePaths.push(targetStoragePath);

      const { data: importedDocument, error: documentInsertError } =
        await supabase
          .from("documents")
          .insert({
            uploader_id: userID,
            workspace_id: null,
            library_id: importedLibrary.id,
            title: sourceDocument.title,
            file_url: targetStoragePath,
            file_size_bytes: sourceDocument.file_size_bytes,
            is_public: false,
            status: "APPROVED",
            ai_reject_reason: null,
          })
          .select("id")
          .single();

      if (documentInsertError) throw documentInsertError;
      insertedDocumentIds.push(importedDocument.id);

      const chunksToInsert = sourceChunks
        .filter((chunk) => String(chunk.document_id) === String(sourceDocument.id))
        .map((chunk) => ({
          document_id: importedDocument.id,
          chunk_index: chunk.chunk_index,
          content: chunk.content,
          embedding: chunk.embedding,
        }));
      const tagsToInsert = sourceTags
        .filter((tag) => String(tag.document_id) === String(sourceDocument.id))
        .map((tag) => ({
          document_id: importedDocument.id,
          tag_id: tag.tag_id,
          assigned_by: userID,
        }));

      if (chunksToInsert.length > 0) {
        const { error } = await supabase
          .from("document_chunks")
          .insert(chunksToInsert);
        if (error) throw error;
      }

      if (tagsToInsert.length > 0) {
        const { error } = await supabase
          .from("document_tags")
          .insert(tagsToInsert);
        if (error) throw error;
      }
    }

    const { error: downloadLogError } = await supabase
      .from("library_downloads")
      .insert({ library_id: sourceLibrary.id });
    if (downloadLogError) {
      console.warn("Could not record imported library download:", downloadLogError);
    }

    return res.status(201).json({
      status: "success",
      data: {
        ...importedLibrary,
        documents: sourceDocuments.length,
        visibility: "private",
        sourceLibraryId: sourceLibrary.id,
      },
    });
  } catch (error) {
    console.error("Import library error:", error);

    if (insertedDocumentIds.length > 0) {
      await supabase.from("document_tags").delete().in("document_id", insertedDocumentIds);
      await supabase.from("document_chunks").delete().in("document_id", insertedDocumentIds);
      await supabase.from("documents").delete().in("id", insertedDocumentIds);
    }
    if (importedLibraryId) {
      await supabase.from("libraries").delete().eq("id", importedLibraryId);
    }
    if (copiedStoragePaths.length > 0) {
      await supabase.storage.from(BUCKET).remove(copiedStoragePaths);
    }

    return res.status(500).json({
      status: "error",
      message: "Could not import this library.",
      error: error.message,
    });
  }
};

Phân tích luồng import:

1. Lấy libraryId từ URL và name/description từ request body.
2. Kiểm tra name bắt buộc và giới hạn 50/350 ký tự.
3. Kiểm tra quyền truy cập thư viện nguồn.
4. Kiểm tra người dùng đã có thư viện cùng tên hay chưa.
5. Tính tổng dung lượng tài liệu được import.
6. Kiểm tra tổng dung lượng Library của người dùng không vượt 50MB.
7. Tạo thư viện mới thuộc người dùng hiện tại.
8. Mặc định thư viện mới là Private và không hiển thị trên profile.
9. Đọc chunks và tags của toàn bộ tài liệu nguồn.
10. Copy từng file trong Supabase Storage sang đường dẫn mới.
11. Tạo document record mới với uploader_id là người import.
12. Sao chép chunks/embedding để AI có thể dùng ngay mà không cần xử lý lại.
13. Sao chép quan hệ tag và đổi assigned_by thành người import.
14. Ghi nhận một lượt download/import cho thư viện nguồn.
15. Trả thư viện mới về frontend.

Rollback:

- copiedStoragePaths ghi lại các file đã copy.
- insertedDocumentIds ghi lại các document đã tạo.
- importedLibraryId ghi lại thư viện mới.
- Nếu một bước thất bại, catch sẽ xóa tag, chunk, document, library và file Storage
  đã tạo trước đó.
- Đây là cơ chế rollback "best effort" do các thao tác database và Storage không nằm
  chung trong một transaction Supabase duy nhất.

function readStorageList(key) {
  try {
    const value = JSON.parse(localStorage.getItem(key) || "[]");
    return Array.isArray(value) ? value : [];
  } catch {
    return [];
  }
}

Lý do xóa:

- Dữ liệu tạo Library hiện được lưu trong Supabase.
- localStorage chỉ chứa dữ liệu trên đúng browser/origin hiện tại.

function resolveLibraryLink() {
  setLinkError("");
  setSourceLibrary(null);
  setSourceItems([]);

  if (!trimmedLink) {
    setLinkError("Please enter a library link.");
    return;
  }

  try {
    const parsedUrl = new URL(trimmedLink, window.location.origin);
    const match = parsedUrl.pathname.match(/\/dashboard\/libraries\/([^/?#]+)/);
    const libraryId = match?.[1];

    if (!libraryId) {
      throw new Error("This is not a valid AI Study Hub library link.");
    }

    const savedLibraries = readStorageList("aiStudyHubLibraries");
    const matchedLibrary = savedLibraries.find(
      (library) => String(library.id) === decodeURIComponent(libraryId),
    );

    if (!matchedLibrary) {
      throw new Error(
        "This library is not available in the current browser session.",
      );
    }

    const importedItems = readStorageList(
      `aiStudyHubImportedLibraryItems:${matchedLibrary.id}`,
    );

    setSourceLibrary(matchedLibrary);
    setSourceItems(importedItems);
    setLibraryName(matchedLibrary.name || "Imported library");
    setDescription(matchedLibrary.description || "");
  } catch (error) {
    setLinkError(error.message || "Cannot read this library link.");
  }
}

async function resolveLibraryLink() {
  setLinkError("");
  setSourceLibrary(null);

  if (!trimmedLink) {
    setLinkError("Please enter a library link.");
    return;
  }

  try {
    const parsedUrl = new URL(trimmedLink, window.location.origin);
    const match = parsedUrl.pathname.match(/\/dashboard\/libraries\/([^/?#]+)/);
    const libraryId = match?.[1];

    if (!libraryId) {
      throw new Error("This is not a valid AI Study Hub library link.");
    }

    setIsCheckingLink(true);
    const matchedLibrary = await previewLibraryImport(
      decodeURIComponent(libraryId),
    );

    setSourceLibrary(matchedLibrary);
    setLibraryName(matchedLibrary.name || "Imported library");
    setDescription(matchedLibrary.description || "");
  } catch (error) {
    setLinkError(
      error.response?.data?.message ||
        error.message ||
        "Cannot read this library link.",
    );
  } finally {
    setIsCheckingLink(false);
  }
}

5. LUỒNG HOẠT ĐỘNG SAU KHI THAY ĐỔI
======================================================================

1. Người dùng dán link:
   /dashboard/libraries/:libraryId

2. Frontend tách libraryId từ URL.

3. Frontend gọi:
   GET /api/documents/libraries/:libraryId/import-preview

4. Backend:
   - Kiểm tra đăng nhập.
   - Tìm thư viện.
   - Kiểm tra Public hoặc chính chủ.
   - Lấy danh sách tài liệu APPROVED.
   - Trả metadata cho giao diện.

5. Người dùng có thể đổi tên và mô tả của bản sao.

6. Frontend gọi:
   POST /api/documents/libraries/:libraryId/import

7. Backend:
   - Xác minh quyền lại lần nữa.
   - Kiểm tra trùng tên.
   - Kiểm tra giới hạn 50MB.
   - Tạo Library Private.
   - Copy file Storage.
   - Copy document records.
   - Copy chunks/embeddings.
   - Copy tags.
   - Trả ID của Library mới.

8. Frontend chuyển đến:
   /dashboard/libraries/:newLibraryId


======================================================================
6. ĐÁNH GIÁ KỸ THUẬT
======================================================================

Điểm đã cải thiện:

- Dữ liệu được lưu thật trong backend thay vì phụ thuộc browser.
- Có thể import giữa hai tài khoản và hai thiết bị khác nhau.
- Có kiểm tra quyền truy cập.
- Không import tài liệu chưa được duyệt.
- Có giới hạn dung lượng.
- Có chống trùng tên cả frontend và backend.
- Bản sao có file Storage riêng, không phụ thuộc file của thư viện nguồn.
- AI chunks và embedding được tái sử dụng, tránh gọi AI lại.
- Có rollback khi quá trình copy thất bại giữa chừng.
- Bản sao mặc định Private để bảo vệ dữ liệu.

Giới hạn hiện tại:

- Folder của LibraryPage vẫn được lưu bằng localStorage nên chưa thể import folder.
- Rollback là best effort, chưa phải transaction database + storage tuyệt đối.
- Quá trình copy đang xử lý file tuần tự; thư viện nhiều file có thể mất thời gian.
- Chưa có share token riêng cho thư viện Private của người khác.
- Chỉ tài liệu APPROVED được sao chép.

Hướng mở rộng đề xuất:

1. Tạo bảng library_folders và document_folder_assignments.
2. Tạo share token có thời hạn cho thư viện Private.
3. Chuyển quy trình import lớn sang background job.
4. Thêm bảng library_imports để theo dõi tiến trình và lịch sử.
5. Thêm idempotency key để tránh hai bản sao khi request bị gửi lại.
6. Đưa thao tác database vào Supabase RPC transaction.
7. Thêm integration test với Supabase test project/storage.
